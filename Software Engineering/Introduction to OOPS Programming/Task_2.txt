#include <iostream>
#include <string>
using namespace std;

class Playlist {
public:
    string name;
    string createdOn;
    bool isPublic;

    Playlist(string n, string date, bool publicStatus) {
        name = n;
        createdOn = date;
        isPublic = publicStatus;
    }

    void togglePublic() {
        isPublic = !isPublic;
    }
};

int main() {
    Playlist playlist("My Favorites", "25-08-2026", true);

    cout << "Initial: "
         << (playlist.isPublic ? "true" : "false") << endl;

    playlist.togglePublic();

    cout << "After first toggle: "
         << (playlist.isPublic ? "true" : "false") << endl;

    playlist.togglePublic();

    cout << "After second toggle: "
         << (playlist.isPublic ? "true" : "false") << endl;

    return 0;
}

#include <iostream>
#include <string>
using namespace std;

class FoodOrder {
public:
    int orderId;
    string restaurantName;
    bool isDelivered;

    FoodOrder(int id, string restaurant, bool delivered) {
        orderId = id;
        restaurantName = restaurant;
        isDelivered = delivered;
    }

    void markDelivered() {
        isDelivered = true;
        cout << "Order " << orderId
             << " has been delivered!" << endl;
    }
};

int main() {
    FoodOrder order(101, "Pizza Hut", false);

    order.markDelivered();

    cout << "Order ID: " << order.orderId << endl;
    cout << "Restaurant: " << order.restaurantName << endl;
    cout << "Delivered: "
         << (order.isDelivered ? "true" : "false") << endl;

    return 0;
}

#include <iostream>
#include <string>
#include <vector>
using namespace std;

class Playlist {
public:
    string name;
    string createdOn;
    bool isPublic;

    vector<string> songs;

    Playlist(string n, string date, bool publicStatus) {
        name = n;
        createdOn = date;
        isPublic = publicStatus;
        songs = {};
    }

    void togglePublic() {
        isPublic = !isPublic;
    }

    void addSong(string songTitle) {
        songs.push_back(songTitle);
    }

    void displaySongs() {
        cout << "Songs:" << endl;

        for (int i = 0; i < songs.size(); i++) {
            cout << i + 1 << ". " << songs[i] << endl;
        }
    }
};

int main() {
    Playlist playlist("My Favorites", "25-08-2026", true);

    playlist.addSong("Perfect");
    playlist.addSong("Believer");
    playlist.addSong("Shape of You");

    playlist.displaySongs();

    return 0;
}

#include <iostream>
#include <string>
using namespace std;

struct FoodOrderData {
    int orderId;
    string restaurantName;
    bool isDelivered;
};

class FoodOrder {
public:
    int orderId;
    string restaurantName;
    bool isDelivered;

    // Constructor takes an object (struct)
    FoodOrder(FoodOrderData data) {
        orderId = data.orderId;
        restaurantName = data.restaurantName;
        isDelivered = data.isDelivered;
    }

    void markDelivered() {
        isDelivered = true;

        cout << "Order " << orderId
             << " has been delivered!" << endl;
    }
};

int main() {
    // Object containing constructor data
    FoodOrderData orderData = {
        101,
        "Pizza Hut",
        false
    };

    // Pass object to FoodOrder constructor
    FoodOrder order(orderData);

    cout << "Order ID: " << order.orderId << endl;
    cout << "Restaurant: " << order.restaurantName << endl;
    cout << "Delivered: "
         << (order.isDelivered ? "true" : "false") << endl;

    order.markDelivered();

    return 0;
}
