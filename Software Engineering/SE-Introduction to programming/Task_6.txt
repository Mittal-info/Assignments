#include <stdio.h>

int main() {
    char teams[10][30] = {
        "Mumbai Indians",
        "Chennai Super Kings",
        "RCB"
    };

    int choice;
    int count = 3;

    while (1) {
        printf("\n1. View Teams\n");
        printf("2. Add New Team\n");
        printf("3. Exit\n");

        printf("Enter choice: ");
        scanf("%d", &choice);

        if (choice == 1) {
            printf("\nFavorite IPL Teams:\n");

            for (int i = 0; i < count; i++) {
                printf("%d. %s\n", i + 1, teams[i]);
            }
        }
        else if (choice == 2) {
            if (count < 10) {
                printf("Enter new team: ");
                scanf(" %[^\n]", teams[count]);

                count++;
                printf("Team added!\n");
            }
        }
        else if (choice == 3) {
            printf("Goodbye!");
            break;
        }
        else {
            printf("Invalid choice!\n");
        }
    }

    return 0;
}

#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <time.h>

int main() {
    char songs[3][30] = {
        "Shape of You",
        "Believer",
        "Perfect"
    };

    char guess[30];
    int randomSong;

    srand(time(0));
    randomSong = rand() % 3;

    do {
        printf("Guess the song: ");
        scanf(" %[^\n]", guess);

        if (strcmp(guess, songs[randomSong]) == 0) {
            printf("Correct! You guessed it!\n");
        } else {
            printf("Wrong guess! Try again.\n");
        }

    } while (strcmp(guess, songs[randomSong]) != 0);

    return 0;
}
