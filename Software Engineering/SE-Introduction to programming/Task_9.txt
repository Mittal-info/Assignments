#include <stdio.h>

float averageSpend(int orders[]) {
    int sum = 0;

    for (int i = 0; i < 7; i++) {
        sum = sum + orders[i];
    }

    return sum / 7.0;
}

int main() {
    int orders[7] = {200, 350, 150, 400, 250, 300, 450};

    printf("Average weekly spend: %.2f", averageSpend(orders));

    return 0;
}

#include <stdio.h>

int main() {
    int cricketScores[3][2] = {
        {180, 175},
        {210, 190},
        {165, 170}
    };

    for (int i = 0; i < 3; i++) {

        if (cricketScores[i][0] > cricketScores[i][1])
            printf("Match %d highest score: %d\n", i + 1, cricketScores[i][0]);
        else
            printf("Match %d highest score: %d\n", i + 1, cricketScores[i][1]);
    }

    return 0;
}
