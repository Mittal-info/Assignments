#include <stdio.h>
#include <ctype.h>

void getUserInitials(char name[], char initials[]) {
    initials[0] = toupper(name[0]);

    for (int i = 0; name[i] != '\0'; i++) {
        if (name[i] == ' ') {
            initials[1] = toupper(name[i + 1]);
            break;
        }
    }

    initials[2] = '\0';
}

int main() {
    char name[] = "Virat Kohli";
    char initials[3];

    getUserInitials(name, initials);

    printf("Initials: %s", initials);

    return 0;
}

#include <stdio.h>
#include <string.h>

void addToCart(char cart[][30], int *count, char product[]) {
    strcpy(cart[*count], product);
    (*count)++;

    printf("Updated Cart:\n");

    for (int i = 0; i < *count; i++) {
        printf("%d. %s\n", i + 1, cart[i]);
    }
}
int main() {
    char cart[10][30] = {"Laptop", "Mouse"};
    int count = 2;

    printf("Before adding:\n");
    for (int i = 0; i < count; i++) {
        printf("%d. %s\n", i + 1, cart[i]);
    }

    addToCart(cart, &count, "Keyboard");

    printf("\nCart outside function:\n");
    for (int i = 0; i < count; i++) {
        printf("%d. %s\n", i + 1, cart[i]);
    }

    return 0;
}

#include <stdio.h>

// Pass by value
void increaseFollowersByValue(int followers) {
    followers = followers + 1000;
    printf("Inside value function: %d\n", followers);
}

// Pass by reference
void increaseFollowersByReference(int *followers) {
    *followers = *followers + 1000;
    printf("Inside reference function: %d\n", *followers);
}

int main() {
    int followers1 = 5000;
    int followers2 = 5000;

    printf("Before value function: %d\n", followers1);
    increaseFollowersByValue(followers1);
    printf("After value function: %d\n\n", followers1);

    printf("Before reference function: %d\n", followers2);
    increaseFollowersByReference(&followers2);
    printf("After reference function: %d\n", followers2);

    return 0;
}

#include <stdio.h>

void formatPrice(int price, char result[]) {
    sprintf(result, "₹%d", price);
}

int main() {
    char price1[20], price2[20], price3[20];

    formatPrice(1599, price1);
    formatPrice(2499, price2);
    formatPrice(999, price3);

    printf("Product 1: %s\n", price1);
    printf("Product 2: %s\n", price2);
    printf("Product 3: %s\n", price3);

    return 0;
}

#include <stdio.h>
#include <ctype.h>

void capitalizeFirst(char text[]) {
    if (text[0] != '\0') {
        text[0] = toupper(text[0]);
    }
}

int main() {
    char product[] = "laptop";
    char username[] = "virat";

    capitalizeFirst(product);
    capitalizeFirst(username);

    printf("Product: %s\n", product);
    printf("Username: %s\n", username);

    return 0;
}
