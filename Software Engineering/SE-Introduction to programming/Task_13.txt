#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main() {
    FILE *file;
    char song[100];

    // 1. Write top 3 songs
    file = fopen("playlist.txt", "w");

    fprintf(file, "Tum Hi Ho\n");
    fprintf(file, "Perfect\n");
    fprintf(file, "Love Story\n");

    fclose(file);

    // 2. Read and display songs
    file = fopen("playlist.txt", "r");

    printf("Songs in playlist:\n");

    while (fgets(song, sizeof(song), file) != NULL) {
        printf("%s", song);
    }

    fclose(file);

    // 3. Append two more songs
    file = fopen("playlist.txt", "a");

    fprintf(file, "What Makes You Beautiful\n");
    fprintf(file, "I Love You\n");

    fclose(file);

    // 4. Find songs containing "love"
    file = fopen("playlist.txt", "r");

    printf("\nSongs containing 'love':\n");

    while (fgets(song, sizeof(song), file) != NULL) {

        // Convert song to lowercase
        for (int i = 0; song[i] != '\0'; i++) {
            song[i] = tolower(song[i]);
        }

        if (strstr(song, "love") != NULL) {
            printf("%s", song);
        }
    }

    fclose(file);

    return 0;
}
