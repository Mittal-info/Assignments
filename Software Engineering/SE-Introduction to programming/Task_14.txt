function isEven(num) {
    // Check whether the number is divisible by 2
    if (num % 2 == 0) {
        return true;
    } else {
        return false;
    }
}

function formatFollowersCount(count) {
    // Numbers below 1000 are returned as they are
    if (count < 1000) {
        return count.toString();
    }

    // Convert thousands into K
    if (count < 1000000) {
        return (count / 1000).toFixed(1) + "K";
    }

    // Convert millions into M
    return (count / 1000000).toFixed(1) + "M";
}

console.log(formatFollowersCount(500));       // 500
console.log(formatFollowersCount(1500));      // 1.5K
console.log(formatFollowersCount(1200000));   // 1.2M
