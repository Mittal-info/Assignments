    <script>

        // Form select karna
        const form = document.getElementById("musicLinksForm");

        const previewCards =
            document.getElementById("previewCards");


        // Form submit hone par
        form.addEventListener("submit", function(event) {

            event.preventDefault();


            // URLs lena
            const spotifyURL =
                document.getElementById("spotify").value;

            const youtubeURL =
                document.getElementById("youtube").value;

            const appleURL =
                document.getElementById("apple").value;


            // Old preview cards remove karna
            previewCards.innerHTML = "";


            // Spotify card
            createPreviewCard(
                "Spotify",
                spotifyURL,
                "bi-spotify",
                "spotify-card"
            );


            // YouTube card
            createPreviewCard(
                "YouTube Music",
                youtubeURL,
                "bi-youtube",
                "youtube-card"
            );


            // Apple Music card
            createPreviewCard(
                "Apple Music",
                appleURL,
                "bi-apple",
                "apple-card"
            );


            // Form clear karna
            form.reset();

        });


        // Preview card create karne ka function

        function createPreviewCard(
            platform,
            url,
            icon,
            cardClass
        ) {

            // URL se username/handle extract karna
            const username = extractUsername(url);


            // New div create karna
            const card = document.createElement("div");

            card.className =
                "preview-card " + cardClass;


            // Card ka HTML
            card.innerHTML = `

                <div class="platform-logo">
                    <i class="bi ${icon}"></i>
                </div>

                <div class="card-info">

                    <h3>${platform}</h3>

                    <p>@${username}</p>

                </div>

                <a href="${url}"
                   target="_blank"
                   class="visit-btn">
                    Visit
                </a>

            `;


            // Card ko page par add karna
            previewCards.appendChild(card);
        }


        // URL se username/handle nikalna

        function extractUsername(url) {

            try {

                const link = new URL(url);

                let path = link.pathname;

                // Starting aur ending slash remove karna
                path = path.replace(/^\/+|\/+$/g, "");


                if (path === "") {
                    return "Music Profile";
                }


                // URL ko parts mein divide karna
                const parts = path.split("/");


                // Last part username hoga
                return parts[parts.length - 1];

            }

            catch {

                return "Music Profile";

            }

        }

    </script>
